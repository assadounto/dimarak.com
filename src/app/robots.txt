// app/robots.txt
import type { MetadataRoute } from 'next';

const SITE = process.env.NEXT_PUBLIC_SITE_URL || 'http://localhost:3000';

export default function robots(): MetadataRoute.Robots {
  return {
    rules: [
      {
        userAgent: '*',
        allow: '/',
        disallow: [
          // Block internal or private paths if needed:
          '/api/',
          '/admin/',
          '/dashboard/',
          '/checkout', // keep checkout out of indexing
          '/cart',     // if you have a /cart route
        ],
      },
    ],
    sitemap: [`${SITE}/sitemap.xml`],
    host: SITE,
  };
}
